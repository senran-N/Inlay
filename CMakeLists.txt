cmake_minimum_required(VERSION 3.24)
project(inlay VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
option(INLAY_BUILD_TESTS "Build Inlay checks" ${PROJECT_IS_TOP_LEVEL})
option(INLAY_BUILD_EXAMPLES "Build document demo" ${PROJECT_IS_TOP_LEVEL})
option(INLAY_INSTALL "Enable packaging" ${PROJECT_IS_TOP_LEVEL})
find_package(Etch 0.16 CONFIG QUIET)
if(NOT TARGET Etch::etch)
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../Etch/CMakeLists.txt")
  set(ETCH_INSTALL ${INLAY_INSTALL})
  add_subdirectory(../Etch etch_build)
 else()
  message(FATAL_ERROR "Inlay requires Etch 0.16 or newer")
 endif()
endif()
function(inlay_cpp target)
 target_compile_features(${target} PUBLIC cxx_std_20)
 set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
 if(MSVC)
  string(REPLACE "/EHsc" "" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}" PARENT_SCOPE)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /GR- /EHs-c- /utf-8)
  target_compile_definitions(${target} PRIVATE _HAS_EXCEPTIONS=0)
 else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions -fno-rtti)
 endif()
endfunction()
add_library(inlay STATIC src/inlay_doc.cpp src/inlay_view.cpp)
add_library(Inlay::inlay ALIAS inlay)
inlay_cpp(inlay)
target_include_directories(inlay PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(inlay PUBLIC $<BUILD_INTERFACE:Etch::etch> $<INSTALL_INTERFACE:Etch::etch>)
if(INLAY_BUILD_TESTS)

  add_test(NAME inlay_downstream_consumers COMMAND ${CMAKE_COMMAND} -DROOT=${CMAKE_CURRENT_SOURCE_DIR} -DBINARY=${CMAKE_CURRENT_BINARY_DIR} -DSDL3_DIR=${SDL3_DIR} -Dharfbuzz_DIR=${harfbuzz_DIR} -DEXE_SUFFIX=${CMAKE_EXECUTABLE_SUFFIX} -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/check_downstream_consumers.cmake)
  set_tests_properties(inlay_downstream_consumers PROPERTIES TIMEOUT 120)

 inlay_cpp(inlay_tests)
 target_link_libraries(inlay_tests PRIVATE Inlay::inlay)

 set_target_properties(inlay_c_abi_smoke PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)

  target_compile_options(inlay_c_abi_smoke PRIVATE /W4 /WX /utf-8)

 add_test(NAME inlay_c_abi_smoke COMMAND inlay_c_abi_smoke)
 if(WIN32)
  foreach(target inlay_tests inlay_c_abi_smoke)
